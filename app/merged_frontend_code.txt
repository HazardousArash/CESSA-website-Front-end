"use client";

import React, { useEffect } from "react";
import Link from "next/link";
import { usePathname, useRouter } from "next/navigation";
import { useAuth } from "@/hooks/useAuth";
import { useDispatch } from "react-redux";
import { clearAuth } from "@/redux/slices/authSlice";
import { cessaService } from "@/lib/api/services/client/cessaApi.service";
import { Home, User, CreditCard, Users, LogOut, ArrowLeft, UsersCore, Terminal, Award, Megaphone } from "lucide-react"; // Import correct icons

export default function DashboardLayout({ children }) {
  const { user, role, isAuthenticated, authChecked } = useAuth();
  const pathname = usePathname();
  const dispatch = useDispatch();
  const router = useRouter();

  useEffect(() => {
    if (authChecked && !isAuthenticated) {
      router.push("/login");
    }
  }, [isAuthenticated, authChecked, router]);

  const handleLogout = async () => {
    try {
      const activeRole = role ? role.toLowerCase() : "student";
      await cessaService.auth.logout(activeRole);
    } finally {
      dispatch(clearAuth());
      router.push("/login");
    }
  };

  if (!authChecked || !isAuthenticated) {
    return <div className="min-h-screen flex items-center justify-center text-cyber-teal font-mono animate-pulse">// SYSTEM_AUTH_VERIFY...</div>;
  }

  // Base navigation
  const navItems = [
    { href: "/dashboard", label: "داشبورد من", icon: Home },
    { href: "/profile", label: "پروفایل کاربری", icon: User },
    { href: "/payments", label: "تراکنش‌های مالی", icon: CreditCard },
  ];

  if (role === "Admin" || role === "ContentCreator") {
    navItems.splice(2, 0, { href: "/explore-users", label: "جستجوی دانشجویان", icon: Users });
  }

  // Break down Admin Menu (#15)
  if (role === "Admin") {
    navItems.push({ href: "/admin/users", label: "مدیریت کاربران", icon: UsersCore });
    navItems.push({ href: "/admin/ads", label: "مدیریت تبلیغات", icon: Megaphone });
    navItems.push({ href: "/admin/halloffame", label: "تالار مفاخر (ادمین)", icon: Award });
    navItems.push({ href: "/admin/logs", label: "لاگ‌های سیستم", icon: Terminal });
  }

  return (
    <div className="min-h-screen relative flex text-gray-200 font-sans z-10">
      <div className="relative z-10 flex w-full h-screen p-4 lg:p-6 gap-6 overflow-hidden max-w-[1600px] mx-auto">
        <aside className="w-72 bg-dark-950/80 backdrop-blur-2xl rounded-[32px] p-6 flex flex-col justify-between shrink-0 shadow-[0_0_40px_rgba(0,0,0,0.5)] border border-white/10 relative overflow-hidden hidden md:flex">
          <div className="absolute top-0 right-0 w-32 h-32 bg-cyber-teal/10 blur-3xl rounded-full pointer-events-none"></div>
          
          <div className="relative z-10 overflow-y-auto pr-2" style={{ scrollbarWidth: 'none' }}>
            <div className="flex flex-col items-center pb-8 pt-4 border-b border-white/5">
              <div className="w-20 h-20 rounded-full bg-dark-900 border-2 border-cyber-teal flex items-center justify-center font-bold text-cyber-teal text-3xl shadow-[0_0_25px_rgba(45,212,191,0.2)]">
                {user?.firstName ? user.firstName[0] : "U"}
              </div>
              <h3 className="font-bold text-white text-lg mt-3 mb-1">
                {user?.firstName ? `${user.firstName} ${user.lastName}` : "کاربر CESSA"}
              </h3>
              <div className="px-3 py-1 rounded-full bg-white/5 border border-white/10 text-[10px] text-cyber-teal font-mono tracking-widest mt-1">
                {role?.toUpperCase() || "STUDENT"}
              </div>
            </div>

            <nav className="mt-8 space-y-2">
              {navItems.map((item) => {
                const Icon = item.icon;
                const isActive = pathname === item.href || pathname.startsWith(item.href + '/');
                return (
                  <Link key={item.href} href={item.href} className={`flex items-center space-x-4 space-x-reverse px-5 py-3.5 rounded-2xl text-sm font-bold transition-all relative group ${isActive ? "bg-cyber-teal text-dark-950 shadow-[0_0_20px_rgba(45,212,191,0.3)]" : "text-gray-400 hover:text-white hover:bg-white/5 border border-transparent"}`}>
                    <Icon size={20} className={isActive ? "text-dark-950" : "text-gray-500 group-hover:text-cyber-teal transition-colors"} />
                    <span>{item.label}</span>
                  </Link>
                );
              })}
            </nav>
          </div>

          <div className="pt-6 border-t border-white/5 space-y-3 relative z-10 mt-auto">
            <Link href="/" className="flex items-center space-x-3 space-x-reverse text-gray-400 hover:text-white px-5 py-3 rounded-2xl text-sm font-bold transition-colors hover:bg-white/5">
              <ArrowLeft size={18} className="text-gray-500" /><span>بازگشت به سایت</span>

  );

"use client";

import React, { useEffect, useState } from "react";

import { Shield, Key, Trash2, Award, Terminal, UploadCloud, Check } from "lucide-react";

export default function AdminPage() {
  const [tab, setTab] = useState("users");
  const [users, setUsers] = useState([]);
  const [logs, setLogs] = useState([]);
  
  const [hofMembers, setHofMembers] = useState([]);
  const [hofForm, setHofForm] = useState({ fullName: "", description: "", githubUrl: "", linkedInUrl: "", telegramId: "", email: "", file: null });

  const loadData = async () => {
    try {
      if (tab === "users") {
        const res = await cessaService.users.adminGetAll({});
        setUsers(res?.data || []);
      } else if (tab === "logs") {
        const res = await cessaService.logs.getSystemLogs();
        setLogs(res?.data || []);
      } else if (tab === "halloffame") {
        const res = await cessaService.hallOfFame.getAll("admin");
        setHofMembers(res?.data || []);
      }
    } catch (err) {
      console.error(err);
    }
  };

  useEffect(() => { loadData(); }, [tab]);

  const handleVerifyCreator = async (id) => {
    if (!confirm("تایید این کاربر به عنوان تولیدکننده محتوا؟")) return;
    try {
      await cessaService.users.adminVerifyCreator(id);
      loadData();
      alert("حساب کاربر با موفقیت تایید شد.");
    } catch (err) {
      alert("خطا در تایید حساب");
    }
  };

  const handleBypassPassword = async (id) => {
    const newPass = prompt("رمز عبور جدید را وارد کنید (حداقل 6 کاراکتر):");
    if (!newPass) return;
    try {
      await cessaService.users.adminBypassPassword(id, newPass);
      alert("رمز عبور کاربر با موفقیت تغییر کرد.");
    } catch (err) {
      alert("خطا در تغییر پسورد");
    }
  };

  const handleRemoveUser = async (id) => {
    if (!confirm("آیا از غیرفعال‌سازی و Soft Delete این کاربر اطمینان دارید؟")) return;
    try {
      await cessaService.users.adminRemove(id);
      loadData();
    } catch (err) {
      alert("خطا در حذف کاربر");
    }
  };

  const handleAddHof = async (e) => {
    e.preventDefault();
    try {
      let photoFileId = null;
      if (hofForm.file) {
        const fileRes = await cessaService.files.upload(hofForm.file);
        photoFileId = fileRes.data;
      }
      
      const payload = {
        fullName: hofForm.fullName,
        description: hofForm.description,
        githubUrl: hofForm.githubUrl,
        linkedInUrl: hofForm.linkedInUrl,
        telegramId: hofForm.telegramId,
        email: hofForm.email,
        photoFileId: photoFileId
      };
      
      await cessaService.hallOfFame.addMember(payload);
      alert("نخبه جدید با موفقیت به تالار مفاخر اضافه شد.");
      setHofForm({ fullName: "", description: "", githubUrl: "", linkedInUrl: "", telegramId: "", email: "", file: null });
      loadData(); 
    } catch (err) {
      alert("خطا در ثبت نخبگان");
    }
  };

  const handleDeleteHof = async (id) => {
    if (!confirm("از حذف این شخص از تالار مفاخر اطمینان دارید؟")) return;
    try {
      await cessaService.hallOfFame.deleteMember(id);
      loadData(); 
    } catch (err) {
      alert("خطا در حذف");
    }
  };

            <span>مرکز فرماندهی مدیریت سیستم</span>

          <p className="text-xs text-gray-400 mt-1 font-mono">// ROOT_SECURITY_CLEARANCE_REQUIRED</p>

          {[{ id: "users", label: "مدیریت کاربران" }, { id: "logs", label: "لاگ‌های سیستم" }, { id: "halloffame", label: "تالار مفاخر" }].map((t) => (
            <button

              className={`px-4 py-2 rounded-xl text-xs font-mono transition-all cursor-pointer ${
                tab === t.id ? "bg-amber-400 text-dark-950 font-bold shadow-[0_0_15px_rgba(251,191,36,0.3)]" : "bg-dark-900 text-gray-400 border border-white/10 hover:border-amber-400/50"
              }`}
            >
              {t.label}
            </button>

        </div>

      }
    };
    verify();
  }, [searchParams]);

    <div className="min-h-screen relative flex flex-col items-center justify-center p-4">

      <div className="relative z-10 glass-panel p-10 rounded-3xl text-center max-w-md w-full border-cyber-teal/30 shadow-[0_0_30px_rgba(45,212,191,0.2)]">
